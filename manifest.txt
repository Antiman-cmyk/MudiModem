# MudiModem manifest — the ONE list of what ships where.
#
# Read by src/installer/mudimodem-install (install, uninstall, keep.d generation), by install.sh
# (which files to fetch), by tools/deploy.sh (which files to pack) and by the tests. Order matters:
# files are placed top to bottom, services are started top to bottom and stopped bottom to top.
#
#   file    <src> <dest> <mode>   copy
#   gz      <src> <dest> <mode>   gzip -9 (page chunks: nginx serves them with gzip_static)
#   cache   <src> <dest> <mode>   gzip, but only when <dest> is not already a readable AT-library
#                                 cache (a refreshed library must survive an update)
#   default <src> <dest> <mode>   copy only when <dest> does not exist (user settings)
#   own     <dir>/                wholly ours: kept across firmware upgrades, removed on uninstall
#   keepd   <path>                where the generated firmware-upgrade keep list goes
#   service <name> <actions>      /etc/init.d/<name> actions after install (comma-separated);
#                                 uninstall runs stop,disable in reverse order
#
# Every <dest> must be under /etc /usr /lib /bin /sbin (abfota_update keeps nothing else) and
# never under /www (a firmware upgrade never restores it). test/manifest.test.sh enforces this and
# that every file under src/ is either listed here or deliberately unshipped.

own     /usr/share/mudimodem/
own     /usr/lib/mudimodem/
own     /etc/mudimodem/
keepd   /lib/upgrade/keep.d/mudimodem
